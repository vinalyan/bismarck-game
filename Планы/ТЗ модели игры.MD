## ТЗ: Модели игры и отображения

### 1. Архитектура данных

#### 1.1. Концепция

Система использует двухуровневую модель данных:
- **GameModel** — полное состояние игры в JSON на сервере (Redis/память)
- **ViewModel** — модель отображения, формируется из GameModel с фильтрацией по стороне игрока

```
PostgreSQL (БД) → GameModel (Redis/память) → ViewModel → Frontend
```

### 4. Сервис управления состоянием

#### 4.1. GameStateService

```go
// internal/game/services/game_state_service.go

package services

import (
    "context"
    "encoding/json"
    "fmt"
    "sync"
    "time"
    
    "bismarck-game/backend/internal/game/models"
    "bismarck-game/backend/pkg/redis"
)

type GameStateService struct {
    redis      *redis.Client
    db         *database.DB
    games      map[string]*models.GameModel // Кэш в памяти
    mutex      sync.RWMutex
    maxHistory int // Максимальное количество снимков в истории (50)
    
    // Зависимости
    unitService      *UnitService
    taskForceService *TaskForceService
    phaseService     *PhaseService
    eventService     *GameEventService
    searchService    *SearchService
    markerService    *HexMarkerService
    mapStructureService *MapStructureService
    wsHub            *WebSocketHub
}

// LoadGameModel - загружает модель игры из Redis/памяти/БД
func (s *GameStateService) LoadGameModel(gameID string) (*models.GameModel, error) {
    // 1. Проверяем память (самый быстрый) - ~0.1ms
    s.mutex.RLock()
    if model, ok := s.games[gameID]; ok {
        s.mutex.RUnlock()
        return model, nil
    }
    s.mutex.RUnlock()
    
    // 2. Проверяем Redis (быстро) - ~1-5ms
    ctx := context.Background()
    redisKey := fmt.Sprintf("game_model:%s", gameID)
    cached, err := s.redis.Get(ctx, redisKey).Result()
    if err == nil {
        var model models.GameModel
        if err := json.Unmarshal([]byte(cached), &model); err == nil {
            s.mutex.Lock()
            s.games[gameID] = &model // Кэшируем в памяти
            s.mutex.Unlock()
            return &model, nil
        }
    }
    
    // 3. Загружаем из БД (медленно) - ~200-500ms (только при первом запросе)
    return s.loadGameModelFromDB(gameID)
}

// loadGameModelFromDB - собирает модель из нормализованных таблиц
func (s *GameStateService) loadGameModelFromDB(gameID string) (*models.GameModel, error) {
    model := &models.GameModel{
        GameID:      gameID,
        Version:     1,
        LastUpdated: time.Now(),
        Units:       make(map[string]*models.UnitModel),
        TaskForces:  make(map[string]*models.TaskForceModel),
        SearchFactors: make(map[string]int),
        HexMarkers:  make(map[string]models.HexMarkersModel),
        Events:      make([]*models.GameEventModel, 0),
        History:     make([]*models.GameModelSnapshot, 0),
    }
    
    // Загружаем юниты
    units := s.unitService.GetUnitsByGameID(gameID)
    for _, unit := range units {
        model.Units[unit.ID] = s.convertUnitToModel(unit)
    }
    
    // Загружаем Task Forces
    taskForces := s.taskForceService.GetTaskForcesByGameID(gameID)
    for _, tf := range taskForces {
        model.TaskForces[tf.ID] = s.convertTaskForceToModel(tf)
    }
    
    // Загружаем текущий ход
    turn := s.phaseService.GetCurrentTurn(gameID)
    model.CurrentTurn = s.convertTurnToModel(turn)
    
    // Загружаем события (последние 100)
    events := s.eventService.GetEventsByGameID(gameID, 100)
    model.Events = s.convertEventsToModel(events)
    
    // Загружаем маркеры
    model.HexMarkers = s.markerService.GetAllMarkers(gameID)
    
    // Загружаем гексы с собственными факторами поиска
    model.IntrinsicSearchHexes = s.mapStructureService.GetIntrinsicSearchHexes()
    
    // Рассчитываем факторы поиска только для релевантных гексов
    playerSide := s.getPlayerSideFromGame(gameID)
    if playerSide != "" {
        relevantHexes := s.searchService.GetRelevantHexesForSearchFactors(model, playerSide)
        for hexID := range relevantHexes {
            factor := s.searchService.CalculateSearchFactorsFromModel(model, hexID, playerSide)
            if factor > 0 {
                model.SearchFactors[hexID] = factor
            }
        }
    }
    
    // Сохраняем в Redis и память
    s.saveGameModel(gameID, model)
    
    return model, nil
}

// UpdateGameModel - обновляет модель игры с сохранением снимка
func (s *GameStateService) UpdateGameModel(
    gameID string,
    actionID string,
    actionType string,
    description string,
    updater func(*models.GameModel) error,
) error {
    s.mutex.Lock()
    defer s.mutex.Unlock()
    
    // Загружаем текущую модель
    model, err := s.LoadGameModel(gameID)
    if err != nil {
        return err
    }
    
    // Создаем снимок ДО изменения
    snapshot := s.createSnapshot(model, actionID, actionType, description)
    
    // Применяем обновление
    if err := updater(model); err != nil {
        return err
    }
    
    // Обновляем версию
    model.Version++
    model.LastUpdated = time.Now()
    
    // Добавляем снимок в историю
    if model.History == nil {
        model.History = make([]*models.GameModelSnapshot, 0)
    }
    model.History = append([]*models.GameModelSnapshot{snapshot}, model.History...)
    
    // Ограничиваем размер истории
    if len(model.History) > s.maxHistory {
        model.History = model.History[:s.maxHistory]
    }
    
    // Пересчитываем факторы поиска только для релевантных гексов
    if model.CurrentTurn.CurrentPhase == "movement" || 
       model.CurrentTurn.CurrentPhase == "search" {
        playerSide := s.getPlayerSideFromGame(gameID)
        if playerSide != "" {
            relevantHexes := s.searchService.GetRelevantHexesForSearchFactors(model, playerSide)
            model.SearchFactors = make(map[string]int)
            for hexID := range relevantHexes {
                factor := s.searchService.CalculateSearchFactorsFromModel(model, hexID, playerSide)
                if factor > 0 {
                    model.SearchFactors[hexID] = factor
                }
            }
        }
    }
    
    // Сохраняем обновленную модель
    s.saveGameModel(gameID, model)
    
    // Сохраняем снимок в БД (асинхронно)
    go s.saveSnapshotToDB(gameID, snapshot)
    
    // Сохраняем изменения в БД (асинхронно)
    go s.persistToDB(gameID, model)
    
    return nil
}

// createSnapshot - создает глубокую копию состояния
func (s *GameStateService) createSnapshot(
    model *models.GameModel,
    actionID string,
    actionType string,
    description string,
) *models.GameModelSnapshot {
    // Глубокая копия через JSON
    snapshotJSON, _ := json.Marshal(model)
    var snapshotModel models.GameModel
    json.Unmarshal(snapshotJSON, &snapshotModel)
    
    return &models.GameModelSnapshot{
        Version:     model.Version,
        Timestamp:   time.Now(),
        ActionID:    actionID,
        ActionType:   actionType,
        Description: description,
        Snapshot:    &snapshotModel,
    }
}

// saveGameModel - сохраняет модель в Redis и память
func (s *GameStateService) saveGameModel(gameID string, model *models.GameModel) {
    modelJSON, _ := json.Marshal(model)
    redisKey := fmt.Sprintf("game_model:%s", gameID)
    ctx := context.Background()
    s.redis.Set(ctx, redisKey, modelJSON, 24*time.Hour)
    
    s.mutex.Lock()
    s.games[gameID] = model
    s.mutex.Unlock()
}

// BuildViewModel - формирует модель отображения из модели игры
func (s *GameStateService) BuildViewModel(
    gameID string,
    playerSide string,
    options *ViewModelOptions,
) (*models.ViewModel, error) {
    gameModel, err := s.LoadGameModel(gameID)
    if err != nil {
        return nil, err
    }
    
    viewModel := &models.ViewModel{
        Success:   true,
        Timestamp: time.Now(),
        CurrentTurn: gameModel.CurrentTurn,
    }
    
    // Фильтруем юниты по видимости для текущего игрока
    for _, unit := range gameModel.Units {
        if s.isUnitVisible(unit, playerSide) {
            viewModel.Units.Units = append(viewModel.Units.Units, 
                s.convertUnitToViewModel(unit))
        }
    }
    
    // Фильтруем Task Forces
    for _, tf := range gameModel.TaskForces {
        if s.isTaskForceVisible(tf, playerSide) {
            viewModel.Units.TaskForces = append(viewModel.Units.TaskForces,
                s.convertTaskForceToViewModel(tf))
        }
    }
    
    // Фильтруем контакты (только актуальные)
    currentTurn := gameModel.CurrentTurn.TurnNumber
    for _, contact := range gameModel.EnemyContacts {
        if contact.Turn >= currentTurn-1 { // Только последние 2 хода
            viewModel.Units.EnemyContacts = append(viewModel.Units.EnemyContacts,
                s.convertContactToViewModel(contact))
        }
    }
    
    // Факторы поиска - только для текущей фазы и релевантных гексов
    if options.IncludeSearch && 
       (gameModel.CurrentTurn.CurrentPhase == "movement" || 
        gameModel.CurrentTurn.CurrentPhase == "search") {
        viewModel.Search.HexFactors = gameModel.SearchFactors
        viewModel.Search.HexMarkers = gameModel.HexMarkers
    }
    viewModel.Search.VisibilityLevel = gameModel.CurrentTurn.VisibilityLevel
    
    // События - только последние N
    limit := options.EventsLimit
    if limit == 0 {
        limit = 15
    }
    events := gameModel.Events
    if len(events) > limit {
        events = events[:limit]
    }
    viewModel.Events.Events = events
    viewModel.Events.Count = len(events)
    viewModel.Events.TotalCount = len(gameModel.Events)
    viewModel.Events.Limit = limit
    if len(events) > 0 {
        viewModel.Events.LastEventID = events[0].ID
    }
    
    return viewModel, nil
}

// ViewModelOptions - опции для формирования ViewModel
type ViewModelOptions struct {
    EventsLimit      int      `json:"events_limit"`
    IncludeSearch    bool     `json:"include_search"`
    HexIDs           []string `json:"hex_ids,omitempty"` // Для фильтрации (опционально)
}

// UndoAction - откатывает последнее действие
func (s *GameStateService) UndoAction(gameID string, userID string) error {
    s.mutex.Lock()
    defer s.mutex.Unlock()
    
    model, err := s.LoadGameModel(gameID)
    if err != nil {
        return err
    }
    
    // Проверяем права
    if !s.canUndoAction(model, userID) {
        return errors.New("cannot undo this action")
    }
    
    // Проверяем наличие истории
    if len(model.History) == 0 {
        return errors.New("no actions to undo")
    }
    
    // Берем последний снимок
    lastSnapshot := model.History[0]
    
    // Восстанавливаем состояние
    restoredModel := lastSnapshot.Snapshot
    restoredModel.Version = model.Version + 1
    restoredModel.LastUpdated = time.Now()
    restoredModel.History = model.History[1:]
    
    // Создаем событие отката
    undoEvent := &models.GameEventModel{
        ID:          uuid.New().String(),
        GameID:      gameID,
        Turn:        restoredModel.CurrentTurn.TurnNumber,
        Phase:       restoredModel.CurrentTurn.CurrentPhase,
        EventType:   "action_undone",
        ActorName:   "System",
        Description: fmt.Sprintf("Откат действия: %s", lastSnapshot.Description),
        Data: map[string]interface{}{
            "undone_action_id":   lastSnapshot.ActionID,
            "undone_action_type": lastSnapshot.ActionType,
            "restored_version":   lastSnapshot.Version,
        },
        CreatedAt: time.Now(),
    }
    restoredModel.Events = append([]*models.GameEventModel{undoEvent}, restoredModel.Events...)
    
    // Сохраняем восстановленную модель
    s.saveGameModel(gameID, restoredModel)
    
    // Сохраняем в БД
    go s.persistToDB(gameID, restoredModel)
    
    // WebSocket уведомление
    s.wsHub.BroadcastToGame(gameID, models.WSMessage{
        Type: models.WSMessageType.GameEvent,
        Data: map[string]interface{}{
            "event": "action_undone",
            "action_id": lastSnapshot.ActionID,
        },
    })
    
    return nil
}

// canUndoAction - проверяет, можно ли откатить действие
func (s *GameStateService) canUndoAction(model *models.GameModel, userID string) bool {
    if len(model.History) == 0 {
        return false
    }
    
    // Можно откатывать только в фазе movement
    if model.CurrentTurn.CurrentPhase != "movement" {
        return false
    }
    
    // Можно откатывать только последние 5 действий
    maxUndoDepth := 5
    if len(model.History) > maxUndoDepth {
        return false
    }
    
    return true
}
```

```go
// GetRelevantHexesForSearchFactors - получает только гексы, где нужно рассчитывать факторы
func (s *SearchService) GetRelevantHexesForSearchFactors(
    gameModel *models.GameModel,
    playerSide string,
) map[string]bool {
    relevantHexes := make(map[string]bool)
    
    // 1. Гексы с кораблями или ТФ игрока
    for _, unit := range gameModel.Units {
        if unit.Nationality == playerSide && 
           unit.Position != "" &&
           unit.Status != "sunk" {
            relevantHexes[unit.Position] = true
        }
    }
    
    for _, tf := range gameModel.TaskForces {
        if tf.Nationality == playerSide && tf.Position != "" {
            relevantHexes[tf.Position] = true
        }
    }
    
    // 2. Гексы с маркерами пути полета
    for hexID, markers := range gameModel.HexMarkers {
        if markers.FlightPathSearch != nil && *markers.FlightPathSearch > 0 {
            relevantHexes[hexID] = true
        }
    }
    
    // 3. Гексы с маркерами морского патруля
    for _, unit := range gameModel.Units {
        if unit.Nationality == playerSide && 
           unit.IsPatrolling &&
           unit.Position != "" {
            relevantHexes[unit.Position] = true
        }
    }
    
    for _, tf := range gameModel.TaskForces {
        if tf.Nationality == playerSide && 
           tf.IsPatrolling &&
           tf.Position != "" {
            relevantHexes[tf.Position] = true
        }
    }
    
    // 4. Гексы с собственными факторами поиска (из конфигурации)
    for hexID := range gameModel.IntrinsicSearchHexes {
        relevantHexes[hexID] = true
    }
    
    return relevantHexes
}

// CalculateSearchFactorsFromModel - расчет из GameModel (без БД)
func (s *SearchService) CalculateSearchFactorsFromModel(
    gameModel *models.GameModel,
    hexID string,
    playerSide string,
) int {
    totalFactors := 0
    
    // 1. Собственные факторы гекса (из конфигурации)
    if intrinsicFactor, ok := gameModel.IntrinsicSearchHexes[hexID]; ok {
        totalFactors += intrinsicFactor
    }
    
    // 2. Корабли и ТФ в гексе
    unitCount := 0
    tfCount := 0
    
    for _, unit := range gameModel.Units {
        if unit.Position == hexID && 
           unit.Nationality == playerSide &&
           unit.TaskForceID == nil &&
           unit.Status != "sunk" &&
           s.canUnitContributeSearchFactors(unit) {
            unitCount++
        }
    }
    
    for _, tf := range gameModel.TaskForces {
        if tf.Position == hexID && tf.Nationality == playerSide {
            tfCount++
        }
    }
    
    totalFactors += unitCount + tfCount
    
    // 3. Маркеры морского патруля (+3 за каждый)
    patrolCount := 0
    for _, unit := range gameModel.Units {
        if unit.Position == hexID && 
           unit.Nationality == playerSide &&
           unit.IsPatrolling {
            patrolCount++
        }
    }
    for _, tf := range gameModel.TaskForces {
        if tf.Position == hexID && 
           tf.Nationality == playerSide &&
           tf.IsPatrolling {
            patrolCount++
        }
    }
    totalFactors += patrolCount * 3
    
    // 4. Маркеры пути полета (+2 за каждый)
    if markers, ok := gameModel.HexMarkers[hexID]; ok {
        if markers.FlightPathSearch != nil {
            totalFactors += *markers.FlightPathSearch * 2
        }
    }
    
    return totalFactors
}

// CalculateSearchFactorsForRelevantHexes - рассчитывает только для релевантных гексов
func (s *SearchService) CalculateSearchFactorsForRelevantHexes(
    gameModel *models.GameModel,
    playerSide string,
) (map[string]int, error) {
    relevantHexes := s.GetRelevantHexesForSearchFactors(gameModel, playerSide)
    
    factors := make(map[string]int)
    
    for hexID := range relevantHexes {
        factor := s.CalculateSearchFactorsFromModel(gameModel, hexID, playerSide)
        if factor > 0 {
            factors[hexID] = factor
        }
    }
    
    return factors, nil
}
```

```go
// POST /api/games/{gameId}/units/{unitId}/move
func (h *MovementHandler) MoveUnit(w http.ResponseWriter, r *http.Request) {
    gameID := vars["gameId"]
    unitID := vars["unitId"]
    
    var req models.MovementRequest
    json.NewDecoder(r.Body).Decode(&req)
    
    // Генерируем ID действия
    actionID := uuid.New().String()
    
    // Загружаем GameModel
    gameModel, err := h.gameStateService.LoadGameModel(gameID)
    if err != nil {
        http.Error(w, "Game not found", http.StatusNotFound)
        return
    }
    
    unit, exists := gameModel.Units[unitID]
    if !exists {
        http.Error(w, "Unit not found", http.StatusNotFound)
        return
    }
    
    fromHex := unit.Position
    description := fmt.Sprintf("%s переместился из %s в %s", unit.Name, fromHex, req.ToHex)
    
    // Выполняем движение с сохранением снимка
    err = h.gameStateService.UpdateGameModel(
        gameID,
        actionID,
        "move",
        description,
        func(model *models.GameModel) error {
            // Обновляем позицию
            unit := model.Units[unitID]
            unit.Position = req.ToHex
            
            // Рассчитываем расход топлива
            fuelCost, _ := h.movementService.CalculateFuelCostFromModel(
                unit, fromHex, req.ToHex, model)
            unit.Fuel -= fuelCost
            unit.LastMoveTurn = model.CurrentTurn.TurnNumber
            unit.MovementUsed = 1
            
            // Создаем событие
            event := &models.GameEventModel{
                ID:          uuid.New().String(),
                GameID:      gameID,
                Turn:        model.CurrentTurn.TurnNumber,
                Phase:       model.CurrentTurn.CurrentPhase,
                EventType:   "unit_moved",
                ActorName:   unit.Name,
                Description: description,
                Data: map[string]interface{}{
                    "action_id":   actionID,
                    "unit_id":     unitID,
                    "from_hex":    fromHex,
                    "to_hex":      req.ToHex,
                    "fuel_cost":   fuelCost,
                    "hexes_moved": 1,
                },
                CreatedAt: time.Now(),
            }
            model.Events = append([]*models.GameEventModel{event}, model.Events...)
            
            return nil
        },
    )
    
    if err != nil {
        http.Error(w, err.Error(), http.StatusInternalServerError)
        return
    }
    
    // WebSocket уведомление
    h.wsHub.BroadcastToGame(gameID, models.WSMessage{
        Type: models.WSMessageType.GameEvent,
        Data: map[string]interface{}{
            "event":     "unit_moved",
            "action_id": actionID,
            "unit_id":   unitID,
            "from_hex":  fromHex,
            "to_hex":    req.ToHex,
        },
    })
    
    response := models.MovementResponse{
        Success:     true,
        Message:     "Unit moved successfully",
        NewPosition: req.ToHex,
        ActionID:    actionID,
        Data: map[string]interface{}{
            "fuel": unit.Fuel,
        },
    }
    
    json.NewEncoder(w).Encode(response)
}
```
